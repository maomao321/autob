# AutoQuant 大模型开仓决策流程图

本流程图对应当前程序中的 ChatGPT、DeepSeek 和双模型决策逻辑。大模型只参与“今日开仓方向”和“候选开仓时机”两道关口；止损、止盈等风险退出不经过大模型审核。

```mermaid
flowchart TD
    START([启动单只股票运行器]) --> SWITCH{"启用大模型决策？"}

    subgraph CONFIG["1. 模式选择与启动校验"]
        SWITCH -- "否" --> MANUAL["使用用户选择的<br/>LONG / SHORT / FLAT"]
        SWITCH -- "是" --> MODE{"模型模式"}
        MODE -- "CHATGPT" --> OKEY{"OpenAI Key<br/>可用？"}
        MODE -- "DEEPSEEK" --> DKEY{"DeepSeek Key<br/>可用？"}
        MODE -- "DUAL" --> BKEY{"两个 Key<br/>都可用？"}
        OKEY -- "否" --> STARTFAIL["拒绝启动并提示缺少凭据"]
        DKEY -- "否" --> STARTFAIL
        BKEY -- "否" --> STARTFAIL
        OKEY -- "是" --> DAILY
        DKEY -- "是" --> DAILY
        BKEY -- "是" --> DAILY
    end

    subgraph DIRECTION["2. 每个交易日决定今日开仓方向"]
        DAILY["收到当日日线 K 线"] --> CONTEXT["采集近期新闻、SPY/QQQ 大盘走势、<br/>标的最近 30 根日线 OHLC 和当日日线"]
        CONTEXT --> DATAOK{"上下文完整且<br/>存在近期新闻？"}
        DATAOK -- "否" --> FLAT["安全降级为 FLAT<br/>当日禁止新开仓"]
        DATAOK -- "是" --> DAILYMODE{"单模型还是 DUAL？"}

        DAILYMODE -- "CHATGPT" --> GPTDAY["OpenAI Responses API<br/>结构化输出方向、置信度、依据和风险"]
        DAILYMODE -- "DEEPSEEK" --> DSDAY["DeepSeek Chat Completions<br/>JSON 输出方向、置信度、依据和风险"]
        DAILYMODE -- "DUAL" --> PARALLEL["并行请求 ChatGPT + DeepSeek"]

        GPTDAY --> SINGLEOK{"响应有效且<br/>置信度达到阈值？"}
        DSDAY --> SINGLEOK
        SINGLEOK -- "否" --> FLAT
        SINGLEOK -- "是" --> DAYDIR["写入今日方向<br/>LONG / SHORT / FLAT"]

        PARALLEL --> CONSENSUS{"两个模型均成功、均达阈值，<br/>并且方向完全一致？"}
        CONSENSUS -- "否" --> FLAT
        CONSENSUS -- "是" --> DAYDIR
        FLAT --> WAITDAY["记录原因与风险<br/>等待下一个交易日"]
    end

    MANUAL --> STREAM
    DAYDIR --> OPENABLE{"方向是<br/>LONG 或 SHORT？"}
    OPENABLE -- "否" --> WAITDAY
    OPENABLE -- "是" --> STREAM

    subgraph SIGNAL["3. 五分钟策略产生候选信号"]
        STREAM["接收 5 分钟行情并完成 MA 预热"] --> EXITCHECK{"当前持仓触发<br/>止损或止盈？"}
        EXITCHECK -- "是" --> EXIT["生成风险退出信号<br/>绕过大模型时机审核"]
        EXITCHECK -- "否" --> CLOSED{"K 线已收盘且<br/>突破条件成立？"}
        CLOSED -- "否" --> NEXT["等待下一次行情更新"]
        CLOSED -- "是" --> CANDIDATE["生成 BUY / SELL<br/>候选开仓信号"]
    end

    subgraph TIMING["4. 大模型审核候选开仓时机"]
        CANDIDATE --> PREFLIGHT{"无未决订单、无重复持仓、<br/>次数/方向/交易状态/信号时效均通过？"}
        PREFLIGHT -- "否" --> BLOCK["阻止下单并记录原因"]
        PREFLIGHT -- "是" --> AISWITCH{"大模型已启用？"}
        AISWITCH -- "否" --> FINALRISK
        AISWITCH -- "是" --> TIMINGCTX["组装时机上下文：今日市场上下文、今日日线 OHLC、<br/>候选原因、最近 60 根五分钟 K 线 OHLC"]
        TIMINGCTX --> TIMINGMODE{"单模型还是 DUAL？"}

        TIMINGMODE -- "CHATGPT / DEEPSEEK" --> SINGLEENTRY["请求 ENTER / WAIT、置信度、<br/>依据和风险"]
        SINGLEENTRY --> ENTEROK{"返回 ENTER 且<br/>置信度达到阈值？"}
        ENTEROK -- "否或异常" --> WAIT["安全降级为 WAIT<br/>放弃本次候选信号"]
        ENTEROK -- "是" --> FRESH

        TIMINGMODE -- "DUAL" --> DUALENTRY["并行请求两个模型的<br/>ENTER / WAIT 决策"]
        DUALENTRY --> BOTHENTER{"两者均成功、均达阈值，<br/>并且都返回 ENTER？"}
        BOTHENTER -- "否" --> WAIT
        BOTHENTER -- "是" --> FRESH{"审核完成后<br/>信号仍在有效期？"}
        FRESH -- "否" --> BLOCK
        FRESH -- "是" --> FINALRISK
    end

    subgraph ORDER["5. 最终风控与订单执行"]
        EXIT --> FINALRISK["计算数量并检查单笔/每日金额限制"]
        FINALRISK --> RESERVE{"SQLite 原子预留<br/>资金和订单成功？"}
        RESERVE -- "否" --> BLOCK
        RESERVE -- "是" --> LASTSTOP{"发送前收到<br/>停止请求？"}
        LASTSTOP -- "是" --> BLOCK
        LASTSTOP -- "否" --> SUBMIT["提交 MARKET 订单"]
        SUBMIT --> RECORD["更新订单生命周期、持仓、均价、<br/>交易次数与运行日志"]
    end

    WAIT --> NEXT
    BLOCK --> NEXT
    RECORD --> NEXT

    classDef start fill:#17324d,color:#fff,stroke:#17324d,stroke-width:2px;
    classDef model fill:#eaf2ff,color:#17324d,stroke:#2563eb,stroke-width:2px;
    classDef decision fill:#fff4df,color:#17324d,stroke:#c56a08,stroke-width:2px;
    classDef safe fill:#e8f7f1,color:#075f4d,stroke:#078a68,stroke-width:2px;
    classDef block fill:#fdecee,color:#9f2530,stroke:#c9363e,stroke-width:2px;
    classDef bypass fill:#f1ecff,color:#5536a8,stroke:#7557c7,stroke-width:2px;

    class START start;
    class GPTDAY,DSDAY,PARALLEL,SINGLEENTRY,DUALENTRY,TIMINGCTX model;
    class SWITCH,MODE,OKEY,DKEY,BKEY,DATAOK,DAILYMODE,SINGLEOK,CONSENSUS,OPENABLE,EXITCHECK,CLOSED,PREFLIGHT,AISWITCH,TIMINGMODE,ENTEROK,BOTHENTER,FRESH,RESERVE,LASTSTOP decision;
    class DAYDIR,CANDIDATE,SUBMIT,RECORD safe;
    class STARTFAIL,FLAT,WAITDAY,WAIT,BLOCK block;
    class EXIT bypass;
```

## 关键安全规则

- 大模型关闭时，系统按手动方向运行，不调用 ChatGPT 或 DeepSeek。
- 大模型开启时，手动方向不作为失败兜底；任何数据缺失、网络异常、格式错误、低置信度或双模型分歧都会转为 `FLAT/WAIT`。
- `DUAL` 模式要求今日方向一致，并要求两个模型都同意 `ENTER`。
- 方向判断固定提供最近 30 根日线，时机判断固定提供最近 60 根五分钟 K 线和今日日线；所有 K 线都包含开盘价、最高价、最低价、收盘价。
- 大模型审核期间若收到停止请求，或审核完成后信号已经过期，系统不会下单。
- 止损和止盈属于风险退出，始终绕过大模型时机审核，避免模型故障阻碍平仓。
- OpenAI/DeepSeek Key 只随启动请求临时传递，或从后端环境变量读取，不写入 `config.json` 和 `running.json`。
