# AutoQuant 量化策略流程图

本流程图对应当前程序的实际执行逻辑，包含行情预热、策略信号、资金风控、订单提交和订单生命周期管理。

```mermaid
flowchart TD
    subgraph START["1. 启动与安全恢复"]
        A["启动单只股票运行器"] --> B["恢复本地订单账本"]
        B --> C["将上次中断的 SUBMITTING 订单标记为 UNKNOWN"]
        C --> D["校验股票代码及当前可交易状态"]
        D --> E["同步未完成的实盘订单"]
        E --> F{"存在 UNKNOWN 实盘订单？"}
        F -- "是" --> LOCK["安全锁定该股票并停止运行"]
        F -- "否" --> WS["连接日线和 5 分钟行情 WebSocket"]
    end

    subgraph MARKET["2. 行情与预热"]
        WS --> G["接收行情更新"]
        G --> STOP{"收到停止请求？"}
        STOP -- "是" --> END["停止策略，不自动平仓"]
        STOP -- "否" --> TYPE{"行情周期"}

        TYPE -- "日线 1d" --> DAILY["更新当日日线"]
        DAILY --> DIR{"日线收盘价与开盘价"}
        DIR -- "收盘价 > 开盘价" --> LONG["方向设为 LONG"]
        DIR -- "收盘价 < 开盘价" --> SHORT["方向设为 SHORT"]
        DIR -- "两者相等" --> FLAT["方向设为 FLAT"]
        LONG --> NEWDAY
        SHORT --> NEWDAY
        FLAT --> NEWDAY
        NEWDAY["如果是新交易日，清空 5 分钟预热数据"] --> G

        TYPE -- "5 分钟" --> EXITCHECK{"程序持仓触发止损或止盈？"}
    end

    subgraph STRATEGY["3. 策略信号判断"]
        EXITCHECK -- "触发" --> EXIT["生成 SELL 风险退出信号"]
        EXITCHECK -- "未触发" --> CLOSED{"5 分钟 K 线已收盘？"}
        CLOSED -- "否" --> G
        CLOSED -- "是" --> VALID["过滤重复、乱序和跨交易日 K 线"]
        VALID --> WARM{"已收到 MA周期 + 1 根 K 线？"}
        WARM -- "否" --> G
        WARM -- "是" --> MA["计算上一时点 MA 和当前 MA"]

        MA --> DIRECTION{"当前日线方向"}

        DIRECTION -- "LONG" --> BUYCOND{"前收盘 ≤ 前MA<br/>当前收盘 > 当前MA<br/>当前收盘 > 前一根最高价？"}
        BUYCOND -- "是" --> BUY["生成 BUY 信号"]
        BUYCOND -- "否" --> G

        DIRECTION -- "SHORT" --> SELLCOND{"前收盘 ≥ 前MA<br/>当前收盘 < 当前MA<br/>当前收盘 < 前一根最低价？"}
        SELLCOND -- "是" --> SELL["生成 SELL 信号"]
        SELLCOND -- "否" --> G

        DIRECTION -- "FLAT 或 UNKNOWN" --> G
    end

    subgraph RISK["4. 下单前资金安全检查"]
        BUY --> COMMON
        SELL --> COMMON
        EXIT --> COMMON

        COMMON{"停止、未决订单或信号过期？"}
        COMMON -- "是" --> BLOCK["阻止本次下单"] --> G
        COMMON -- "否" --> SIDE{"信号方向"}

        SIDE -- "BUY" --> BUYRISK["检查已有程序持仓<br/>当日入场次数<br/>单笔金额上限<br/>账户每日买入上限"]
        SIDE -- "SELL" --> SELLRISK["实盘必须存在程序多头<br/>使用全部程序持仓数量<br/>禁止建立空头"]

        BUYRISK --> PASS{"检查通过？"}
        SELLRISK --> PASS
        PASS -- "否" --> BLOCK
        PASS -- "是" --> TRADABLE["实盘再次校验当前交易方向是否可用"]
        TRADABLE --> RESERVE["SQLite 事务原子预留资金及订单"]

        RESERVE --> ATOMIC{"存在未知或未决订单、重复持仓、超量卖出？"}
        ATOMIC -- "是" --> BLOCK
        ATOMIC -- "否" --> LASTSTOP{"发送前再次收到停止请求？"}
        LASTSTOP -- "是" --> CANCELLOCAL["本地标记 REJECTED，不发送"] --> END
        LASTSTOP -- "否" --> ORDER["提交 MARKET 订单"]
    end

    subgraph LIFECYCLE["5. 订单生命周期"]
        ORDER --> MODE{"交易模式"}
        MODE -- "PAPER" --> PAPER["按参考价记录模拟成交"]
        PAPER --> UPDATE["更新程序持仓、均价、交易次数和日买入金额"]
        UPDATE --> G

        MODE -- "REAL" --> RESULT{"Binance 返回结果"}
        RESULT -- "明确拒绝" --> REJECT["记录 REJECTED，释放资金预留"] --> G
        RESULT -- "网络超时或结果不确定" --> UNKNOWN["记录 UNKNOWN"] --> LOCK
        RESULT -- "已接受" --> QUERY["立即查询订单详情"]

        QUERY --> STATUS{"订单状态"}
        STATUS -- "NEW 或 PARTIALLY_FILLED" --> PENDING["记录未决状态，阻止后续订单并定期查询"] --> G
        STATUS -- "FILLED、CANCELED 或 EXPIRED" --> UPDATE
        STATUS -- "状态无法解析或成交数据缺失" --> UNKNOWN
    end
```

## 核心规则

- 正常入场信号只使用已经收盘的 5 分钟 K 线。
- 止损和止盈会随每次 5 分钟行情更新检查，触发后优先覆盖普通策略信号。
- 实盘为只做多模式：`BUY` 建仓，`SELL` 只能平掉程序确认的多头。
- 点击停止只阻止新订单，不会撤销未决订单，也不会自动平仓。
- 任何无法确认状态的实盘订单都会进入安全锁定。
